login block-for 120 attempts 3 within 60

 permit 10.0.99.0 0.0.0.255

line vty 0 4
 logging synchronous
 access-class MGMT-SSH in
 transport input ssh
 transport output none
 login local
 exit

vlan 20
 name Servers
vlan 99
 name Management
vlan 999

exit

no ip dhcp snooping information option

ip arp inspection vlan 10,20,99

 ip arp inspection trust
 no shutdown
