conf t

hostname BR-ASW1
no ip domain-lookup

line con 0
logging synchronous
exit

service password-encryption
enable secret 123

username admin privilege 15 secret 123
ip domain-name lab.local
crypto key generate rsa modulus 2048
ip ssh version 2

ip ssh time-out 60
ip ssh authentication-retries 3
login block-for 120 attempts 3 within 60

ip access-list standard MGMT-SSH
 permit 10.1.99.0 0.0.0.255
 permit 10.0.99.0 0.0.0.255
 deny any log

line vty 0 4
 logging synchronous
 access-class MGMT-SSH in
 transport input ssh
 transport output none
 login local
 exit

vlan 10
 name HR
vlan 20
 name Finance
vlan 99
 name Management
vlan 999
 name NATIVE-BLACKHOLE
 exit

spanning-tree mode rapid-pvst
spanning-tree vlan 10,99 root primary
spanning-tree vlan 20 root secondary

ip dhcp snooping
ip dhcp snooping vlan 10,20,99
no ip dhcp snooping information option

ip arp inspection vlan 10,20,99

interface g0/2
 description Uplink-to-BR-CORE1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,99,999
 switchport nonegotiate
 spanning-tree portfast disable
 ip dhcp snooping trust
 ip arp inspection trust
 no shutdown

interface g0/0
 description Link-to-BR-ASW2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,99,999
 switchport nonegotiate
 spanning-tree portfast disable
 ip dhcp snooping trust
 ip arp inspection trust
 no shutdown

interface g0/1
 description HR-Section
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation restrict
 switchport port-security mac-address sticky
 ip arp inspection trust
 no shutdown

interface vlan 99
 ip address 10.1.99.11 255.255.255.0
 no shutdown
 exit

ip default-gateway 10.1.99.1

exit
